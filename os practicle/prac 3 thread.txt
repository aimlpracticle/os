#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>
#include <time.h>

#define BUFFER_SIZE 3
#define TOTAL_ITEMS 5

int buffer[BUFFER_SIZE];
int in = 0, out = 0;

pthread_mutex_t mutex;
sem_t empty, full;

void* producer(void* arg)
{
    int thread_id = *(int*)arg;

    for (int i = 0; i < TOTAL_ITEMS; i++)
    {
        int item = rand() % 100 + 1;

        sem_wait(&empty);
        pthread_mutex_lock(&mutex);

        buffer[in] = item;

        printf("[PRODUCER %d] Inserted Item %d at Slot %d\n",
               thread_id, item, in);

        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&full);

        sleep(1);
    }

    pthread_exit(NULL);
}

void* consumer(void* arg)
{
    int thread_id = *(int*)arg;

    for (int i = 0; i < TOTAL_ITEMS; i++)
    {
        sem_wait(&full);
        pthread_mutex_lock(&mutex);

        int item = buffer[out];

        printf("[CONSUMER %d] Extracted Item %d from Slot %d\n",
               thread_id, item, out);

        out = (out + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&empty);

        sleep(2);
    }

    pthread_exit(NULL);
}

int main()
{
    pthread_t prod_thread, cons_thread;
    int p_id = 1, c_id = 1;

    srand(time(NULL));

    pthread_mutex_init(&mutex, NULL);
    sem_init(&empty, 0, BUFFER_SIZE);
    sem_init(&full, 0, 0);

    pthread_create(&prod_thread, NULL, producer, &p_id);
    pthread_create(&cons_thread, NULL, consumer, &c_id);

    pthread_join(prod_thread, NULL);
    pthread_join(cons_thread, NULL);

    pthread_mutex_destroy(&mutex);
    sem_destroy(&empty);
    sem_destroy(&full);

    printf("Execution completed successfully.\n");

    return 0;
}



//Compile and run on Ubuntu
Save the program as producer.c, then run:
gcc producer.c -o producer -pthread
./producer
