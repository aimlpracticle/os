#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

#define MAX 100

void bubble_sort(int arr[], int n)
{
    int i, j, temp;

    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - i - 1; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

void selection_sort(int arr[], int n)
{
    int i, j, min_idx, temp;

    for (i = 0; i < n - 1; i++)
    {
        min_idx = i;

        for (j = i + 1; j < n; j++)
        {
            if (arr[j] < arr[min_idx])
                min_idx = j;
        }

        temp = arr[min_idx];
        arr[min_idx] = arr[i];
        arr[i] = temp;
    }
}

void print_array(int arr[], int n)
{
    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);

    printf("\n");
}

int main()
{
    int arr[MAX], child_arr[MAX], parent_arr[MAX];
    int n, choice;

    printf("Enter number of integers to sort: ");
    scanf("%d", &n);

    if (n <= 0 || n > MAX)
    {
        printf("Invalid array size.\n");
        return 1;
    }

    printf("Enter %d integers: ", n);

    for (int i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
        child_arr[i] = arr[i];
        parent_arr[i] = arr[i];
    }

    printf("\n1. Normal Execution\n");
    printf("2. Demonstrate Zombie Process\n");
    printf("3. Demonstrate Orphan Process\n");
    printf("Enter choice: ");
    scanf("%d", &choice);

    if (choice < 1 || choice > 3)
    {
        printf("Invalid choice.\n");
        return 1;
    }

    pid_t pid = fork();

    if (pid < 0)
    {
        perror("Fork failed");
        exit(1);
    }
    else if (pid == 0)
    {
        // Child Process
        printf("\n[Child] PID: %d, Parent PID: %d\n",
               getpid(), getppid());

        if (choice == 3)
        {
            printf("[Child] Sleeping for 5 seconds...\n");
            sleep(5);

            printf("[Child] New Parent PID: %d\n", getppid());
        }

        printf("[Child] Sorting using Selection Sort...\n");
        selection_sort(child_arr, n);

        printf("[Child] Sorted array: ");
        print_array(child_arr, n);

        printf("[Child] Exiting.\n");
        exit(0);
    }
    else
    {
        // Parent Process
        printf("\n[Parent] PID: %d, Child PID: %d\n",
               getpid(), pid);

        printf("[Parent] Sorting using Bubble Sort...\n");
        bubble_sort(parent_arr, n);

        printf("[Parent] Sorted array: ");
        print_array(parent_arr, n);

        if (choice == 1)
        {
            printf("[Parent] Waiting for child...\n");
            wait(NULL);
            printf("[Parent] Child completed successfully.\n");
        }
        else if (choice == 2)
        {
            printf("[Parent] Sleeping without calling wait().\n");
            printf("Check zombie status using another terminal.\n");

            sleep(10);

            printf("[Parent] Calling wait() to reap child.\n");
            wait(NULL);
        }
        else if (choice == 3)
        {
            printf("[Parent] Exiting immediately.\n");
            exit(0);
        }
    }

    return 0;
}
