#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <fcntl.h>
#include <sys/wait.h>

#define BUFFER_SIZE 1024

int main()
{
    int pipe1[2];
    int pipe2[2];
    pid_t pid;

    if (pipe(pipe1) == -1)
    {
        perror("Pipe1 creation failed");
        exit(EXIT_FAILURE);
    }

    if (pipe(pipe2) == -1)
    {
        perror("Pipe2 creation failed");
        exit(EXIT_FAILURE);
    }

    pid = fork();

    if (pid < 0)
    {
        perror("Fork failed");
        exit(EXIT_FAILURE);
    }

    if (pid > 0)
    {
        // Parent Process
        close(pipe1[0]);
        close(pipe2[1]);

        char filepath[BUFFER_SIZE];
        char buffer[BUFFER_SIZE];
        ssize_t bytes_read;

        printf("Enter file pathname: ");
        fgets(filepath, sizeof(filepath), stdin);
        filepath[strcspn(filepath, "\n")] = '\0';

        write(pipe1[1], filepath, strlen(filepath) + 1);
        close(pipe1[1]);

        wait(NULL);

        printf("\n--- File Content ---\n");

        while ((bytes_read = read(pipe2[0], buffer,
                                  sizeof(buffer) - 1)) > 0)
        {
            buffer[bytes_read] = '\0';
            printf("%s", buffer);
        }

        printf("\n--------------------\n");
        close(pipe2[0]);
    }
    else
    {
        // Child Process
        close(pipe1[1]);
        close(pipe2[0]);

        char filepath[BUFFER_SIZE];
        char buffer[BUFFER_SIZE];
        ssize_t bytes_read;

        bytes_read = read(pipe1[0], filepath,
                          sizeof(filepath) - 1);

        if (bytes_read > 0)
        {
            filepath[bytes_read] = '\0';

            int fd = open(filepath, O_RDONLY);

            if (fd < 0)
            {
                char *msg = "Error: Cannot open the file.";
                write(pipe2[1], msg, strlen(msg));
            }
            else
            {
                ssize_t n;

                while ((n = read(fd, buffer,
                                 sizeof(buffer))) > 0)
                {
                    write(pipe2[1], buffer, n);
                }

                close(fd);
            }
        }

        close(pipe1[0]);
        close(pipe2[1]);

        exit(EXIT_SUCCESS);
    }

    return 0;
}
